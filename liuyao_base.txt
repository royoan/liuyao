@dataclass
class FiveElements:
    """
    五行生克关系封装类
    核心功能：生克判断、旺衰计算、五行对应转换、基础刑冲合害判定
    """
    element: str
    # 基础配置：五行顺序、生克关系、旺相休囚死（按四季）
    ELEMENTS: ClassVar[List[str]] = ["木", "火", "土", "金", "水"]
    GENERATION: ClassVar[Dict[str, str]] = {"木": "火", "火": "土", "土": "金", "金": "水", "水": "木"}  # 生
    RESTRAINT: ClassVar[Dict[str, str]] = {"木": "土", "火": "金", "土": "水", "金": "木", "水": "火"}  # 克
    # 五行对应：地支/四季/方位（可扩展）
    BRANCH_MAP = {
        "木": ["寅", "卯"],
        "火": ["巳", "午"],
        "土": ["辰", "丑", "未", "戌"],
        "金": ["申", "酉"],
        "水": ["子", "亥"]
    }
    SEASON_MAP = {"木": "春", "火": "夏", "土": "四季", "金": "秋", "水": "冬"}
    # 旺相休囚死评分（越高越旺）
    PROSPERITY_SCORE = {"旺": 3, "相": 2, "休": 1, "囚": 0, "死": -1}

    def __post_init__(self):
        """
        初始化五行对象
        :param element: 五行（木/火/土/金/水），大小写不敏感
        """
        if self.element not in self.ELEMENTS:
            raise ValueError(f"无效五行：{self.element}，仅支持{self.ELEMENTS}")

    def generates(self, target: str) -> bool:
        """
        判断本五行是否生目标五行
        :param target: 目标五行
        :return: True/False
        """
        if target not in self.ELEMENTS:
            raise ValueError(f"无效五行：{target}")
        return self.GENERATION[self.element] == target

    def restrains(self, target: str) -> bool:
        """
        判断本五行是否克目标五行
        :param target: 目标五行
        :return: True/False
        """
        if target not in self.ELEMENTS:
            raise ValueError(f"无效五行：{target}")
        return self.RESTRAINT[self.element] == target

    def is_generated_by(self, target: str) -> bool:
        """
        判断本五行是否被目标五行生
        :param target: 目标五行
        :return: True/False
        """
        return FiveElements(target).generates(self.element)

    def is_restrained_by(self, target: str) -> bool:
        """
        判断本五行是否被目标五行克
        :param target: 目标五行
        :return: True/False
        """
        return FiveElements(target).restrains(self.element)

    def get_prosperity_status(self, season: str) -> str:
        """
        根据季节判断五行旺相休囚死状态
        五行随四季（春、夏、秋、冬）更替，呈现五种状态。六爻中土的规则特殊，需单独记忆。
        <pre>
        表格
        状态	定义	核心逻辑
        旺	当令得时	季节的主宰，能量最强（如木在春）。
        相	辅佐得势	被旺者所生（如火在春，木生火），能量次强。
        休	退休卸任	生旺者（如水在春，水生木），能量退隐，平和。
        囚	被克受制	克旺者（如金在春，金克木），被旺气反制，能量最弱。
        死	被泄无气	旺者所克（如土在春，木克土），能量被泄尽。
        1. 五行四季旺衰总表（六爻实战标准）  
        这是六爻断卦的绝对标准，请以此为准进行编码。
        表格
        五行	春季 (木旺)	夏季 (火旺)	秋季 (金旺)	冬季 (水旺)	四季末 (土旺)
        木	     旺	        休	        死	        相	      囚
        火	     相	        旺	        囚	        死	      休
        土	     死	        相	        休	        囚	      旺
        金	     囚	        死	        旺	        休	      相
        水	     休	        囚	        相	        旺	      死
        2. 关键特殊规则（编程必看）
        土的特殊性：土不主某一个完整季节，而是主四季的最后 18 天（辰、未、戌、丑月）。在六爻量化中，为了简化计算，通常将辰、戌、丑、未这四个月直接判定为 “土旺”。
        三合局优先：若卦中形成三合局（如申子辰水局），局中五行的旺衰会被局气加持，优先级高于季节旺衰。
        </pre>

        :param season: 季节（春/夏/四季/秋/冬）
        :return: 旺/相/休/囚/死
        """
        if season not in self.SEASON_MAP.values():
            raise ValueError(f"无效季节：{season}，仅支持{list(self.SEASON_MAP.values())}")
        
        # 旺：当令；相：生我者；休：我生者；囚：克我者；死：我克者
        season_element = [k for k, v in self.SEASON_MAP.items() if v == season][0]
        if self.element == season_element:
            return "旺"
        elif self.is_generated_by(season_element):
            return "相"
        elif self.generates(season_element):
            return "休"
        elif self.is_restrained_by(season_element):
            return "囚"
        else:
            return "死"

    def get_prosperity_score(self, season: str) -> int:
        """
        获取五行在指定季节的旺衰得分
        :param season: 季节
        :return: 得分（3/2/1/0/-1）
        """
        status = self.get_prosperity_status(season)
        return self.PROSPERITY_SCORE[status]

    def get_branches(self) -> list:
        """
        获取本五行对应的地支列表
        :return: 地支列表（如木→["寅","卯"]）
        """
        return self.BRANCH_MAP[self.element]

    def is_combination(self, target_branch: str) -> bool:
        """
        基础六合判定（仅判断同五行合，扩展可加地支六合）
        :param target_branch: 目标地支
        :return: True/False
        """
        # 同五行地支为合（如寅卯合木，申酉合金）
        return target_branch in self.get_branches()

    def is_clash(self, target_branch: str) -> bool:
        """
        基础六冲判定（五行相冲）
        :param target_branch: 目标地支
        :return: True/False
        """
        # 先获取目标地支的五行
        target_element = [k for k, v in self.BRANCH_MAP.items() if target_branch in v][0]
        # 五行相冲：木克土、土克水、水克火、火克金、金克木（反向也冲）
        return self.restrains(target_element) or self.is_restrained_by(target_element)


@dataclass
class EarthlyBranch:
    """
    十二地支核心类（修正版）
    核心修复：三合局由「三会」改为六爻标准「长生-帝旺-墓库」组合
    功能：冲/合/刑/害全判定、五行归属、季节对应、月份映射、三合局批量判断等
    """
    #构造函数必填参数
    branch: str

    _element: str | None = None
    _month: str | None = None
    _month_season: str | None = None
    # 基础配置：十二地支全量列表
    BRANCHES = ["子", "丑", "寅", "卯", "辰", "巳", "午", "未", "申", "酉", "戌", "亥"]
    
    # 1. 地支→五行映射
    BRANCH_TO_ELEMENT = {
        "子": "水", "丑": "土", "寅": "木", "卯": "木",
        "辰": "土", "巳": "火", "午": "火", "未": "土",
        "申": "金", "酉": "金", "戌": "土", "亥": "水"
    }
    
    # 2. 地支六合（成对，双向）
    COMBINATION_PAIRS = [("子", "丑"), ("寅", "亥"), ("卯", "戌"), 
                         ("辰", "酉"), ("巳", "申"), ("午", "未")]
    
    # 3. 地支六冲（成对，双向）
    CLASH_PAIRS = [("子", "午"), ("丑", "未"), ("寅", "申"), 
                   ("卯", "酉"), ("辰", "戌"), ("巳", "亥")]
    
    # 4. 地支三刑（分类型：无礼刑/恃势刑/无恩刑/自刑）
    PUNISH_PAIRS = [
        ("寅", "巳"), ("巳", "申"), ("申", "寅"),  # 恃势刑（寅巳申三刑）
        ("丑", "戌"), ("戌", "未"), ("未", "丑"),  # 无恩刑（丑戌未三刑）
        ("子", "卯"), ("卯", "子"),                # 无礼刑（子卯相刑）
        ("辰", "辰"), ("午", "午"), ("酉", "酉"), ("亥", "亥")  # 自刑
    ]
    
    # 5. 地支六害（成对，双向）
    HARM_PAIRS = [("子", "未"), ("丑", "午"), ("寅", "巳"), 
                  ("卯", "辰"), ("申", "亥"), ("酉", "戌")]
    
    # ✅ 核心修复：六爻标准三合局（长生-帝旺-墓库）
    # 格式：{五行: (长生, 帝旺, 墓库)}
    TRIAD_COMBINATIONS = {
        "水": ("申", "子", "辰"),  # 申子辰合水局
        "火": ("寅", "午", "戌"),  # 寅午戌合火局
        "木": ("亥", "卯", "未"),  # 亥卯未合木局
        "金": ("巳", "酉", "丑"),  # 巳酉丑合金局
        "土": None  # 三合局无独立土局，土以四库论（辰戌丑未）
    }

    # 6. 地支→月份映射（农历，1-12月）
    BRANCH_TO_MONTH = {
        "寅": 1, "卯": 2, "辰": 3, "巳": 4, "午": 5, "未": 6,
        "申": 7, "酉": 8, "戌": 9, "亥": 10, "子": 11, "丑": 12
    }
    
    # 7. 地支→季节映射（含土的特殊规则：四季末月）
    BRANCH_TO_SEASON = {
        "寅": "春", "卯": "春", "辰": "春",  # 辰为春末土
        "巳": "夏", "午": "夏", "未": "夏",
        "申": "秋", "酉": "秋", "戌": "秋",
        "亥": "冬", "子": "冬", "丑": "冬"
    }
    
    # 8. 季节→当令五行（用于旺衰判断）
    SEASON_TO_ELEMENT = {"春": "木", "夏": "火", "秋": "金", "冬": "水", "四季": "土"}

    # 9. 五行 在十二地支的 十二长生 状态
    # 十二长生阶段
    STAGES = ["长生", "沐浴", "冠带", "临官", "帝旺",
              "衰", "病", "死", "墓", "绝", "胎", "养"]
    # 五行长生起点
    # 土为什么不长生在巳？ 记住“火土同源，生旺同步”
    # 金为什么不长生在土（丑辰未戌）？巳虽属火，但其内藏天干为‌丙（火）、戊（土）、庚（金）‌，其中庚金为金之本气，故巳火并非纯火，而是“火中含金”，可煅炼金成器。 金气始于夏初（巳月），此时阳气渐盛、火土相生，为金提供“待时而发”的生机，而非盛夏（午月）的极端燥热
    # 丑辰未戌 属四墓库，功能是收、藏、止；代表五行气的终结或归宿，而非初生。
    ZHANGSHENG_START = {
        "木": "亥",
        "火": "寅",
        "土": "寅",  #为什么不长生在巳？ 记住“火土同源，生旺同步”
        "金": "巳",  
        "水": "申"
    }

    # 10. 地支→化变类型, 化进神
    HUAJIN_TYPE: ClassVar[dict[str, str]] = {
        "寅": "卯",
        "巳": "午",
        "申": "酉",
        "子": "亥",
        "丑": "辰", #丑 → 辰
        "辰": "未", # 辰 → 未
        "未": "戌", # 未 → 戌
    }


    def __post_init__(self):
        """__init__函数后， 严格校验输入"""
        if self.branch not in self.BRANCHES:
            raise ValueError(f"无效地支：{self.branch}，仅支持{self.BRANCHES}")
        
        # 缓存核心属性
        self.element = self.BRANCH_TO_ELEMENT[self.branch]
        self.month = self.BRANCH_TO_MONTH[self.branch]
        self.season = self.BRANCH_TO_SEASON[self.branch]
        self.season_element = self.SEASON_TO_ELEMENT[self.season]

    # ========== 基础属性获取 ==========
    def get_element(self) -> str:
        """获取当前地支的五行归属"""
        return self.element

    def get_month(self) -> int:
        """获取当前地支对应的农历月份（1-12）"""
        return self.month

    def get_season(self) -> str:
        """获取当前地支对应的季节（春/夏/秋/冬）"""
        return self.season

    def _validate_target_branch(self, target_branch: str) -> None:
        """私有辅助：校验目标地支合法性"""
        if target_branch not in self.BRANCHES:
            raise ValueError(f"无效目标地支：{target_branch}，仅支持{self.BRANCHES}")

    # ========== 冲合刑害判定 ==========
    def is_combined_with(self, target_branch: str) -> bool:
        """判断当前地支是否与目标地支六合"""
        self._validate_target_branch(target_branch)
        pair = (self.branch, target_branch)
        return pair in self.COMBINATION_PAIRS or pair[::-1] in self.COMBINATION_PAIRS

    def is_clashed_with(self, target_branch: str) -> bool:
        """判断当前地支是否与目标地支六冲"""
        self._validate_target_branch(target_branch)
        pair = (self.branch, target_branch)
        return pair in self.CLASH_PAIRS or pair[::-1] in self.CLASH_PAIRS

    def is_punished_with(self, target_branch: str) -> bool:
        """判断当前地支是否与目标地支相刑（含三刑/无礼刑/自刑）"""
        self._validate_target_branch(target_branch)
        return (self.branch, target_branch) in self.PUNISH_PAIRS or (target_branch, self.branch) in self.PUNISH_PAIRS

    def is_harmed_with(self, target_branch: str) -> bool:
        """判断当前地支是否与目标地支六害"""
        self._validate_target_branch(target_branch)
        pair = (self.branch, target_branch)
        return pair in self.HARM_PAIRS or pair[::-1] in self.HARM_PAIRS

    # ========== 三合局核心逻辑（修正版） ==========
    def get_triad_combination(self) -> Optional[Tuple[str, str, str]]:
        """获取当前地支所属的标准三合局，无则返回None"""
        for triad in self.TRIAD_COMBINATIONS.values():
            if triad and self.branch in triad:
                return triad
        return None

    def is_in_triad(self, target_branches: List[str]) -> bool:
        """判断当前地支与目标地支是否能组成完整三合局"""
        for b in target_branches:
            self._validate_target_branch(b)
        
        current_triad = self.get_triad_combination()
        if not current_triad:
            return False
        
        # 必须包含三合局的全部三支
        all_branches = {self.branch} | set(target_branches)
        return set(current_triad).issubset(all_branches)

    def get_wangxiangxiuqiusi(self, wuxing) -> str:
        """获取参数 五行 在本地支月对应季节的旺衰状态（旺/相/休/囚/死）"""
        #当令者旺(相同)，我生者相，生我者休，克我者囚，我克者死
        month_wuxing = FiveElements(self.element)
        if month_wuxing.element == wuxing:
            return "旺"
        elif month_wuxing.generates(wuxing):
            return "相"
        elif month_wuxing.is_generated_by(wuxing):
            return "休"
        elif month_wuxing.restrains(wuxing):
            return "囚"
        else:
            return "死"


    # ========== 旺衰判定（兼容降级） ==========
    def get_prosperity_status(self) -> str:
        """获取当前地支在对应季节的旺衰状态（旺/相/休/囚/死）
        
        """
        try:
            self_elem = FiveElements(self.element)
            season_elem = FiveElements(self.season_element)
            
            if self.element == self.season_element:
                return "旺"
            elif self_elem.is_generated_by(season_elem.element):
                return "相"
            elif self_elem.generates(season_elem.element):
                return "休"
            elif self_elem.is_restrained_by(season_elem.element):
                return "囚"
            else:
                return "死"
        except ImportError:
            # 本地降级实现
            generation = {"木": "火", "火": "土", "土": "金", "金": "水", "水": "木"}
            restraint = {"木": "土", "火": "金", "土": "水", "金": "木", "水": "火"}
            
            if self.element == self.season_element:
                return "旺"
            elif generation.get(self.season_element) == self.element:
                return "相"
            elif generation.get(self.element) == self.season_element:
                return "休"
            elif restraint.get(self.season_element) == self.element:
                return "囚"
            else:
                return "死"

    # ========== 公开类方法（修正版） ==========
    @classmethod
    def get_all_branches(cls) -> List[str]:
        """获取所有十二地支列表"""
        return cls.BRANCHES

    @classmethod
    def get_branches_by_element(cls, element: str) -> List[str]:
        """根据五行获取对应的地支列表"""
        if element not in cls.SEASON_TO_ELEMENT.values():
            raise ValueError(f"无效五行：{element}，仅支持{list(cls.SEASON_TO_ELEMENT.values())}")
        return [b for b, e in cls.BRANCH_TO_ELEMENT.items() if e == element]

    @classmethod
    def is_triad_combination(cls, branches: List[str]) -> bool:
        """
        类方法：严格判断地支列表是否构成标准三合局
        规则：必须包含且仅需包含某三合局的全部三支（顺序无关）
        """
        # 校验输入合法性
        for b in branches:
            if b not in cls.BRANCHES:
                raise ValueError(f"无效地支：{b}，仅支持{cls.BRANCHES}")
        
        branch_set = set(branches)
        # 遍历所有标准三合局，判断是否完全匹配
        for triad in cls.TRIAD_COMBINATIONS.values():
            if triad and set(triad) == branch_set:
                return True
        return False

    @classmethod
    def get_triad_type(cls, branches: List[str]) -> Optional[str]:
        """返回三合局对应的五行，非三合局则返回None"""
        if not cls.is_triad_combination(branches):
            return None
        
        branch_set = set(branches)
        for elem, triad in cls.TRIAD_COMBINATIONS.items():
            if triad and set(triad) == branch_set:
                return elem
        return None

    @classmethod
    def get_zhangsheng_stage(cls, branch: str, element: str) -> str:
        """返回五行 在地支的 十二长生 阶段"""
        if element not in cls.ZHANGSHENG_START:
            raise ValueError("无效五行")

        start_branch = cls.ZHANGSHENG_START[element]

        current_index = cls.BRANCHES.index(branch)
        start_index = cls.BRANCHES.index(start_branch)

        stage_index = (current_index - start_index) % 12

        return cls.STAGES[stage_index]
    @classmethod
    def is_hua_jin(cls, dongyao_branch: str, bianyao_branch: str) -> bool:
        """判断 动爻 跟 变爻 是否是化进神 """
        return cls.HUAJIN_TYPE.get(dongyao_branch) == bianyao_branch
    
    @classmethod
    def is_hua_tui(cls, dongyao_branch: str, bianyao_branch: str) -> bool:
        """判断 动爻 跟 变爻 是否是化退神 """
        return cls.HUAJIN_TYPE.get(bianyao_branch) == dongyao_branch


@dataclass
class GuaYao():
    '''
    重卦中的每一爻的信息封装类
    '''
    yao_index: int  # 爻序号(从下往上, 从1开始)
    yinyang: int  # 爻的阴阳(1表示阳爻，0表示阴爻)
    liuqin: str  # 父母，兄弟，子孙，妻财，官鬼
    branch: str  # 地支
    element: str  # 五行

    shiyao: bool = field(init=False, default=False)      #是否 世爻
    yingyao: bool = field(init=False, default=False)    #是否 应爻
    yongshen: bool = field(init=False, default=False)   #是否 用神

        # self.__init_name()

        # self.__init_guagong()

    def __load_bugua_yaml_file(self):
        yaml_file_path = 'bagua.yaml'
        with open(yaml_file_path, "r", encoding="utf-8") as f:
            data = yaml.safe_load(f)

        for palace_data in data:
            for palace_name, palace_content in palace_data.items():
                # 验证每卦
                for idx, gua in enumerate(palace_content["八卦"]):
                    if self.guayao_str == gua['六爻']:
                        self.name = gua['名称']
                        self.guagong = palace_name
                        self.guagong_yingyang = palace_content['阴阳']
                        self.guagong_element = palace_content['五行']
                        self.liuqin = gua['六亲']
                        self.dizhi = gua['地支']
                        self.wuxing = gua['五行']
                        self.shiyao = gua['世爻']
                        self.yingyao = gua['应爻']
                        break 
        # 初始化每一爻对象
        for i in range(1, 7):
            guayao = GuaYao(i, int(self.guayao_str[i - 1]), self.liuqin[i - 1], self.dizhi[i - 1], self.wuxing[i - 1])
            if i in self.dongyao_list:
                guayao.dongyao = 1
            if i == self.yingyao:
                guayao.yingyao = 1
            if i == self.shiyao:
                guayao.shiyao = 1
            self.yaos.append(guayao)

    def get_yao(self, yao_index: int) -> GuaYao:
        '''
        获取指定爻的字符串表示
        :param yao_index: 爻序号(从下往上, 从1开始)
        '''
        return self.yaos[yao_index - 1]

    def print_guayao(self):
        for i in range(6, 0, -1):
            print(i)
            print(self.get_yao_str(i))

    def get_yao_str(self, yao_index: int) -> str:
        '''
        获取指定爻的字符串表示
        :param yao_index: 爻序号(从下往上, 从1开始)
        '''
        if yao_index < 1 or yao_index > 6:
            raise ValueError("爻序号必须在1到6之间")
        return self.get_yao(yao_index).get_yao_str()

class ForceInterface(ABC):
    '''
    定义作用接口， 作用来源，作用类型，作用名称，作用力量
    '''
    @abstractmethod
    def get_source(self) -> any:
        """获取作用来源， 一般为月建，日辰，变爻，动爻，静爻
        如果是爻， 就会返回爻的对象"""
        pass
    
    @abstractmethod
    def get_type(self) -> str:
        """获取作用类型,一般为 生克冲合刑害， 四时旺相， 十二长生"""
        pass

    @abstractmethod
    def get_name(self) -> str:
        """获取作用名称,例如月建克， 日建合绊"""
        pass
    @abstractmethod
    def get_score(self) -> float:
        """获取作用力量得分"""
        pass

@dataclass
class MonthForce(ForceInterface):
    '''
    月建 作用力量类型 生克冲合刑害, 本质是地支间的 “互动关系”
    '''
    yue_branch: str  # 月建 地支
    yao_branch: str  # 爻 地支
    #下面的都是私有属性
    _month_element: str = field(init=False) # 月建五行
    _yao_element: str = field(init=False, default="")  # 月建五行

    type: str = field(init=False, default="月建生克冲合刑害")

    name: str | None = None  # 作用类型名称：
    score: float = 0.0  # 作用力量：0.0-10.0

    # 类属性
    # 作用类型名称
    MON_FORCE_TYPE: ClassVar[dict[str, float]] = {
        "临月建":   5.0,    # 地支相同, 力量最强（得令）
        "月破":     -5.0,     # 地支相冲
        "月建生合": 4.5,   # 地支相合， 且五行生爻的五行
        "月建克合": -2.5,   # 地支克合， 且五行克爻的五行, 克中带合，衰而不死（比纯克轻）
        "月建合起": 3.0,     # 地支相合，爻如果 休囚无气，就是合起，  判 “休囚 / 衰弱”：爻无月日生扶、四时休囚 / 死、十二长生为墓 / 绝
        "月建合绊": -1.5,     # 地支相合，爻如果旺向有用， 就是合绊， 判 “旺相 / 有用”：爻得月日生扶、四时旺 / 相、十二长生为长生 / 帝旺，或本身是动爻
        "月建生":   3.5,     
        "月建比和": 4.0,     # 五行相同但地支不同（如寅月卯木），帮扶力较弱 
        "月建克":   -3.0,
        "月建刑":   -2.0,
        "月建害":   -1.0,
        "月建无作用": 0.0,
    }

    def __post_init__(self):
        '''
        计算月建对爻的 生克冲合 作用及力量得分
        '''
        self._month_element = EarthlyBranch(self.yue_branch).element
        self._yao_element = EarthlyBranch(self.yao_branch).element
        
        if self.yue_branch == self.yao_branch:
            self.name = "临月建"
        elif EarthlyBranch(self.yue_branch).is_clashed_with(self.yao_branch):
            self.name = "月破"
        elif EarthlyBranch(self.yue_branch).is_combined_with(self.yao_branch) and FiveElements(self._month_element).generates(self._yao_element):
            self.name = "月建生合"
        elif EarthlyBranch(self.yue_branch).is_combined_with(self.yao_branch) and FiveElements(self._month_element).restrains(self._yao_element):
            self.name = "月建克合"
        elif FiveElements(self._month_element).generates(self._yao_element):
            self.name = "月建生"
        elif FiveElements(self._month_element).restrains(self._yao_element):
            self.name = "月建克"
        elif self._month_element == self._yao_element:
            self.name = "月建比和"  # 又叫比和， 
        elif EarthlyBranch(self.yue_branch).is_combined_with(self.yao_branch):
            self.name = "月建合起"  # 这里要算完所有的爻的旺衰状态， 再判断是否是合起，还是合绊
        elif EarthlyBranch(self.yue_branch).is_punished_with(self.yao_branch):
            self.name = "月建刑"
        elif EarthlyBranch(self.yue_branch).is_harmed_with(self.yao_branch):
            self.name = "月建害"
        else:
            self.name = "月建无作用"
        self.score = self.MON_FORCE_TYPE.get(self.name, 0.0)

    def get_source(self):
        return "月建" + self.yue_branch
    
    def get_name(self):
        return self.name
    def get_score(self):
        return self.score
    def get_type(self):
        return self.type


    @classmethod
    def force(cls, month_branch: str, yao_branch: str) -> float:
        """根据月建地支和爻的地支，返回作用力量"""
        pass

    dongyao: GuaYao  # 动爻
    source: str = "变爻"
    type: str = "变爻生克冲合刑害"
    #下面的都是私有属性
    _bianyao_branch: str = field(init=False, default="") # 变爻 地支
    _dongyao_branch: str = field(init=False, default="") # 动爻 地支

    _bianyao_element: str = field(init=False, default="") # 日建五行
    _dongyao_element: str = field(init=False, default="")  # 爻五行

    _bianyao_force_coeff: float = field(init=False, default=0.0)  # 变爻 作用系数 (范围是0-1， 就是把变爻的总力量映射到这个区间)
    _xunkong_coeff: float = field(init=False, default=0.0)  # 旬空系数 (不空是1， 假空0.8, 真空0.2)

    name: str | None = None  # 作用类型名称：例如化回头生 等
    score: float = 0.0  # 作用力量：0.0-10.0

    # 动爻化变类型及得分， 另外 动爻五行 在变爻的地支 所处的十二长生另外算
    BIAN_FORCE_TYPE: ClassVar[dict[str, float]] = {
        "化回头生":    4.5,     # 
        "化进神":      4.0,       # 
        "化回头生合":  3.0,     # 
        "化回头克":   -4.5,
        "化回头克合": -3.0,     # 
        "化回头合":    2.0,     # 这种要注意，如果动爻旺相有气，就是合绊，减少动爻的力量; 如果动爻本身休囚无气，就是合起，增加动爻的力量，
        "化退神":     -4.0,
        "化回头冲":   -3.5,
        "化回头刑":   -2.0,
        "化回头害":   -1.0,
        "化回头无作用": 0.0
    }

    # 旬空系数
    XUNKONG_COEFF: ClassVar[dict[str, float]] = {
        "不空": 1.0,
        "假空": 0.8,
        "真空": 0.0,
    }

    def __post_init__(self):
        #先计算变爻的月日及旬空得分(范围[-13,13])， 然后把得分变成一个 有权系数范围[0-1]
        # F月 + F月四时+ F月十二长生 + F日 + F日十二长生 总分范围是 [-13,13]
        self._bianyao_force_coeff = self.bianyao.score / 26 + 0.5 # 从[-13,13] 映射到 [0,1]
        
        # 判断化变类型
        if FiveElements(self.bianyao.element).generates(self.dongyao.element) and EarthlyBranch(self.bianyao.branch).is_combined_with(self.dongyao.branch):
            self.name = "化回头生合"
        elif FiveElements(self.bianyao.element).restrains(self.dongyao.element) and EarthlyBranch(self.bianyao.branch).is_combined_with(self.dongyao.branch):
            self.name = "化回头克合"
        if FiveElements(self.bianyao.element).generates(self.dongyao.element):
            self.name = "化回头生"
        elif FiveElements(self.bianyao.element).restrains(self.dongyao.element):
            self.name = "化回头克"
        elif EarthlyBranch.is_hua_jin(self.dongyao.branch, self.bianyao.branch):
            self.name = "化进神"
        elif EarthlyBranch.is_hua_tui(self.dongyao.branch, self.bianyao.branch):
            self.name = "化退神"
        elif EarthlyBranch(self.bianyao.branch).is_combined_with(self.dongyao.branch):
            self.name = "化回头合"
        elif EarthlyBranch(self.bianyao.branch).is_clashed_with(self.dongyao.branch):
            self.name = "化回头冲"
        elif EarthlyBranch(self.bianyao.branch).is_punished_with(self.dongyao.branch):
            self.name = "化回头刑"
        elif EarthlyBranch(self.bianyao.branch).is_harmed_with(self.dongyao.branch):
            self.name = "化回头害"
        else:
            self.name = "化回头无作用"

        self.score = self.BIAN_FORCE_TYPE.get(self.name, 0.0) * self._bianyao_force_coeff * ChangeForce.XUNKONG_COEFF[self.bianyao.xunkong_type]
        # 计算几种特殊的类型：
        # 如果变爻 被 日或月合住，就不会克动爻，这个时候分数为0
        if ('合' in self.bianyao.get_month_force().get_name() or '合' in self.bianyao.get_day_force().get_name()) \
                and self.name == "化回头克":
            self.name = "化回头克，但是贪合忘克"
            self.score = 0.0

    def get_source(self):
        return self.source
    def get_name(self):
        return self.name
    def get_score(self):
        return self.score
    def get_type(self):
        return self.type

@dataclass
class DongYaoForce(ForceInterface):
    '''
    动爻对 本卦的其他爻的作用力类型和大小。
    动爻对用神的核心作用为生、克、冲、合、刑、害六大类，并结合化回头生 / 克、化进 / 化退等变爻关系，形成完整作用体系。
    作用力类型：生克冲合， 作用力大小：0.0-10.0
    '''
    #构造函数
    dongyao: GuaYao  # 动爻
    targetyao: GuaYao  # 目标爻 (可能是 用神，世爻，应爻)
    
    source: str = "动爻"
    type: str = "动爻生克冲合刑害"
    #下面的都是私有属性
    _dongyao_branch: str = field(init=False, default="") # 变爻 地支
    _target_branch: str = field(init=False, default="") # 动爻 地支
    _dongyao_element: str = field(init=False, default="") # 日建五行
    _target_element: str = field(init=False, default="")  # 爻五行

    # 动爻生克冲合得分，合 > 冲 > 生 > 比和 > 克 > 刑 > 害
    DONG_FORCE_TYPE: ClassVar[dict[str, float]] = {
        "动生合":  4.0,     # 又生又合
        "动支相助": 4.0,  # 地支相同
        "动生":    3.0,     # 
        "动平合":  2.0,     # 无生无克只合；牵绊、绑定，吉带束缚
        "动比和":  2.0,     # （五行相同帮扶）
        "动害":   -1.0,     # 子未 / 丑午 / 寅巳 / 卯辰 / 申亥 / 酉戌相害； 主猜忌、小人、暗阻、小事不顺
        "动刑":   -2.0,     # 
        "动克":   -3.0,       # 
        "动克合": -4.0,     #  比克更持久
        "动冲":   -4.0,     # 
        "动无作用": 0.0
    }

    def __post_init__(self):
        self.source = self.dongyao.name + "动爻"
        self._dongyao_branch = self.dongyao.branch
        self._target_branch = self.targetyao.branch
        self._dongyao_element = self.dongyao.element
        self._target_element = self.targetyao.element

        if EarthlyBranch(self._dongyao_branch).is_combined_with(self._target_branch):
            if FiveElements(self._dongyao_element).generates(self._target_element):
                self.name = "动生合"
            elif FiveElements(self._dongyao_element).restrains(self._target_element):
                self.name = "动克合"
            else:
                self.name = "动平合"
        elif self._dongyao_branch == self._target_branch:
            self.name = "动支相助"
        elif self._dongyao_element == self._target_element:
            self.name = "动比和"
        elif EarthlyBranch(self._dongyao_branch).is_harmed_with(self._target_branch):
            self.name = "动害"
        elif EarthlyBranch(self._dongyao_branch).is_punished_with(self._target_branch):
            self.name = "动刑"
        elif FiveElements(self._dongyao_element).restrains(self._target_element):
            self.name = "动克"
        elif EarthlyBranch(self._dongyao_branch).is_clashed_with(self._target_branch):
            self.name = "动冲"
        else:
            self.name = "动无作用"
        
        # F月 + F月四时+ F月十二长生 + F日 + F日十二长生 + 动爻 总分范围是 [-13,13]
        self._dongyao_force_coeff = self.dongyao.score / 26 + 0.5 # 从[-13,13] 映射到 [0,1]

        self.score = self.DONG_FORCE_TYPE.get(self.name, 0.0) * self._dongyao_force_coeff * ChangeForce.XUNKONG_COEFF[self.dongyao.xunkong_type]


    def get_source(self):
        return self.source
    def get_name(self):
        return self.name
    def get_score(self):
        return self.score
    def get_type(self):
        return self.type
