@dataclass
class App:
    # 构造函数参数属性
    qigua_date: str # 起卦时间公历 格式为: yyyy-MM-dd
    guayao_str: str
    dongyao_list: list[int]
    yongshen: str  # 用神, 例如：妻财，或数字，直接指定爻的序号(1-6)
    wenshi: str

    #私有属性
    _bengua: ChongGua = field(init=False, default=None)
    _biangua: ChongGua = field(init=False, default=None)
    _year_ganzhi: str = field(init=False)
    _month_ganzhi: str = field(init=False)
    _month_branch: str = field(init=False)
    _month_element: str = field(init=False)
    _day_ganzhi: str = field(init=False)
    _day_branch: str = field(init=False)
    _day_element: str = field(init=False)
    _season: str = field(init=False)
    _day_xunkong: str = field(init=False) #日旬空 两个连续的日地支，例如: 戌亥

    yongshen_yao: GuaYao = field(init=False, default=None)

    def __post_init__(self):
        self.__init_time_ganzhi(self.qigua_date)
        self.__init_bengua()
        self.__init_biangua()
        self.__calculate_each_yao_month_day_force() #计算每个爻的日月及受力作用，是否旬空
        # 判断 暗动之爻， 日冲旺相有气的静爻
        self.__calculate_bianyao_true_false_empty() #计算每个变爻 是 真空还是假空

        self.__calculate_dongyao_bianyao_force() #计算每个动爻的化变的受力作用

        # self.__calculate_dongyao_month_day_force() # 重新计算每个动爻的综合受力作用
        self.__init_yongshen()


    def __init_time_ganzhi(self, qigua_date: str):
        self.qigua_year = int(qigua_date.split('-')[0]) #年
        self.qigua_month = int(qigua_date.split('-')[1]) #月建
        self.qigua_day = int(qigua_date.split('-')[2]) #日建
        
        self.time = GanZhiTime(self.qigua_year, self.qigua_month, self.qigua_day)

        self._year_ganzhi = self.time.getYearInGanZhi() 
        self._month_ganzhi = self.time.getMonthInGanZhi()  #月建
        self._day_ganzhi = self.time.getDayInGanZhi() #日建
        self._day_xunkong = self.time.getDayXunKong() #旬空 两个日地支
        self._month_branch = self._month_ganzhi[1]  #月 地支
        self._day_branch = self._day_ganzhi[1]      #日 地支
        self._month_element = EarthlyBranch(self._month_branch).get_element()  #月 五行
        self._day_element = EarthlyBranch(self._day_branch).get_element()      #日 五行
        self._season = EarthlyBranch(self._month_branch).get_season()  #根据月地支返回季节 春夏秋冬    

    def __init_bengua(self):
        self._bengua = ChongGua(self.guayao_str, self.dongyao_list)
        # 设置世应对象
        for idx, yao in enumerate(self._bengua.yaos):
            if yao.shiyao == 1:
                self.shiyao = yao
            elif yao.yingyao == 1:
                self.yingyao = yao

    def __init_biangua(self):
        if self.dongyao_list:
            _guayao_list = list(self.guayao_str)
            for dy_index in self.dongyao_list:
                _guayao_list[dy_index - 1] = str(abs(int(_guayao_list[dy_index - 1]) - 1))
            self._biangua = ChongGua(''.join(_guayao_list), None)
        # 设置动爻的变爻对象
        for idx, yao in enumerate(self._bengua.yaos):
            if yao.dongyao:
                yao.bianyao = self._biangua.yaos[idx]

    def __calculate_each_yao_month_day_force(self):
        '''计算每个爻的日月及受力作用'''
        for yao in self._bengua.yaos:
            yao.add_force(MonthForce(self._month_branch, yao.branch))
            yao.add_force(MonthSishiForce(self._month_branch, yao.branch))
            yao.add_force(ZhangshengForce(self._month_branch, yao.branch))
            yao.add_force(DayForce(self._day_branch, yao.branch))
            yao.add_force(ZhangshengForce(self._day_branch, yao.branch, source='日辰'))
            
            yao.xunkong = yao.branch in self._day_xunkong #是否旬空

            if yao.dongyao:
                bianyao = yao.bianyao
                bianyao.add_force(MonthForce(self._month_branch, bianyao.branch))
                bianyao.add_force(MonthSishiForce(self._month_branch, bianyao.branch))
                bianyao.add_force(ZhangshengForce(self._month_branch, bianyao.branch))
                bianyao.add_force(DayForce(self._month_branch, bianyao.branch))
                bianyao.add_force(ZhangshengForce(self._month_branch, bianyao.branch))
                bianyao.xunkong = bianyao.branch in self._day_xunkong #是否旬空

    def __calculate_bianyao_true_false_empty(self):
        '''
        计算每个变爻的真空假空。 真空之变爻 永远不产生回头生、克、合、冲、进退等作用，变爻作用力固定 = 0，无法影响动爻。
        1. 变爻假空（有力，只是暂时不发）
            满足任意一条就是假空：
            变爻临月建
            变爻临日建
            变爻得月生
            变爻得日生
            变爻与月比和
            变爻与日比和
        2. 变爻真空（彻底无力）
            满足全部：
            变爻休囚无气
            被日月克
            无生扶
            月破 或 日破
        '''
        for yao in self._bengua.yaos:
            if yao.dongyao:
                bianyao = yao.bianyao
                month_force = bianyao.get_month_force()  # 月建 score >=3.5 分就满足，临月建，生，生合，比和
                day_force = bianyao.get_day_force()  # 日建 score >=3.5 分就满足，临日建，生，生合，比和
                if (month_force.score >= 3.5 or day_force.score >= 3.5):
                    bianyao.xunkong_type = "假空"
                else:
                    bianyao.xunkong_type = "真空"


    def __calculate_dongyao_bianyao_force(self):
        '''计算每个动爻受变爻受力作用 和 动变类型'''
        dongyaos = [yao for yao in self._bengua.yaos if yao.dongyao]
        for yao in dongyaos:
            bianyao = yao.bianyao
            yao.add_force(ChangeForce(bianyao, yao))  # 变爻的生克冲合刑害
            yao.add_force(ZhangshengForce(bianyao.branch, yao.branch, source='变爻')) # 变爻的十二长生，例如化墓，化绝

    def analyze_yongshen(self):
        '''分析用神 受日月，动爻， 是否有变爻的影响'''
        if not self.yongshen_yao:
            raise ValueError("没有用神， 无法分析用神")
        # 先计算用神是否 旬空， 真空还是假空（动不为空，有
        self.calc_yongshen_xunkong()

        #先计算动爻对用神的影响
        for yao in self._bengua.yaos:
            if yao.dongyao and yao.yao_index != self.yongshen_yao.yao_index:
                self.yongshen_yao.add_force(DongYaoForce(yao, self.yongshen_yao)) # 动爻对其他爻的作用力量：这里简化只考虑 生克冲合， 怎么计算？

        return self.yongshen_yao.score 

    def calc_yongshen_xunkong(self):
        '''
        计算用神是否 旬空， 真空还是假空
        增删：旺不为空；动不为空；有日建动爻动扶者，亦不为空；动而化空，伏而旺相皆不为空。月破为空，有气不动亦为空，伏而被克亦为空，真空为空。真空者，春土夏金秋是木，三冬逢火是真空。
        '''
        if self.yongshen_yao.branch in self._day_xunkong:
            # 旬空
            self.yongshen_yao.set_xunkong(True)
            if self.yongshen_yao.get_month_force().get_name() == "月破":
                # 月破， 旬空为 真空
                self.yongshen_yao.set_xunkong_type('真空')
            elif self.yongshen_yao.get_score() > 5:
                self.yongshen_yao.set_xunkong_type('假空') #TODO 假空条件待完善
            else:
                self.yongshen_yao.set_xunkong_type('真空')
        else:
            self.yongshen_yao.set_xunkong(False)

    def print_guahua(self):
        '''打印整个卦画， 含所测事项，起卦干支，旬空，本卦，变卦'''
        guahua_str = []
        # 第一行
        wenshi_str = f"所测事项：{self.wenshi}"
        guahua_str.append(wenshi_str)
        # 第二行
        qigua_ganzhi_str = f"起卦时间：{self._year_ganzhi}年 {self._month_ganzhi}月 {self._day_ganzhi}日   旬空({self._day_xunkong})  公历：{self.qigua_date} "
        guahua_str.append(qigua_ganzhi_str)
        # 第三行
        qigua_ganzhi_next_str = f"                   {self._month_element}     {self._day_element}"
        guahua_str.append(qigua_ganzhi_next_str)
        # 第四行
        if self.dongyao_list:
            gua_name_str = f"    本卦：{self._bengua.name}       变卦：{self._biangua.name}"
        else:
            gua_name_str = f"    本卦：{self._bengua.name}"
        guahua_str.append(gua_name_str)
    
        # 第五行  六爻
        for i in range(6, 0, -1):
            if self.dongyao_list:
                guahua_str.append(f"{self._bengua.get_yao_str(i)} {self._biangua.get_yao_str(i)}")
            else:
                guahua_str.append(f"{self._bengua.get_yao_str(i)}")

        print("\n".join(guahua_str))
