# 四时旺相系数
# 旺、相、休、囚、死是对事物的性质的形容。在四时中，由于寒、暖、燥、湿不同， 五行 也会受到彩响，发生旺衰起伏的变化，这样的变化状态被称为旺、相、休、死、囚。
# 当令者旺，我生者相，生我者休，克我者囚，我克者死
# 例如寅月，卯月 对应五行为木， 所以木 旺于 春，辰月为土， 木克土为（克我者囚），所以木在辰月 为囚
# 所以 五行的 四时（季）旺相休囚死 只要判断 月地支 跟该 五行的 生克 即可。

        self.qigua_year = int(qigua_date.split('-')[0]) #年

    def _calc_month_score(self, guayao: GuaYao) -> (float, list[str]):
        """
        计算月建作用力量（修正版）：区分临月建、月生、月克、月合，符合六爻经典规则
        1.分层权重体系
            临月建（1.2）：爻与月建同地支（如寅月寅木），为 “得令”，力量最强。
            月建生用神（1.0）：月建五行生用神五行（如寅月午火），为 “得相”，次旺。
            月建合用神（0.9）：月建与用神六合（如寅月亥水），力量略弱于相生。
            同类帮扶（0.8）：五行相同但地支不同（如寅月卯木），帮扶力较弱。
            月建克用神（-1.0）：月建五行克用神五行（如寅月申金），为 “休囚”，力量为负。
            月破（-1.5）：月建冲用神（如寅月申金），为 “月破”，力量最弱。
        2.优先级逻辑
            先判月破（最高优先级，直接返回）→ 再判临月建 → 再判月合 → 最后判五行生克，符合六爻断卦的优先级规则。
        3.修正系数优化
            空亡修正：月建空亡则力量减半（系数 0.5），符合 “空亡则力弱” 的原则。
            四时系数：改为(1 + si_shi_coeff)，使四时旺相系数作为增量调整，而非直接相乘，更符合实际旺衰影响（如春季木旺，si_shi_coeff=0.2，总系数为 1.2）。
        
        params:
        guayao: 爻对象 (一般是用神的爻，从1开始)
        """
        # # 基础权重配置（严格按力量层级划分）
        # YUE_JIAN_WEIGHTS = {
        #     "临月建": 1.2,    # 爻与月建同地支，力量最强（得令）     地支相同
        #     "月建生用神": 1.0, # 月建五行生用神五行，次旺（得相）    五行生
        #     "月建合用神": 0.9, # 月建与用神地支六合，力量略弱于相生   地支六合
        #     "同类帮扶": 0.8,   # 五行相同但地支不同（如寅月卯木），帮扶力较弱    五行相同
        #     "无作用": 0.0,     # 无生克冲合关系
        #     "月建克用神": -1.0, # 月建五行克用神五行，力量为负; 用神力量被削弱，但并非完全失效。若用神本身有日辰生扶、动爻生助或化回头生，仍可转危为安。性质‌：是“被克制”，属于常态性衰弱，有“月克日生”等化解可能。 五行克
        #     "月破": -1.5       # 月建冲用神（月破）地支相冲，力量最弱; 月破被视为“枯根朽木”，能量被彻底冲散，‌在当月内几乎失去作用‌。即使有日辰生扶，也难起效，除非是“假破”（如动爻生扶、化回头生）。‌性质‌：是“被冲散”，属于‌极衰之象‌，是六爻中公认的凶兆之一。《增删卜易》等经典指出：“用神临月破，倒霉之象，虽有似无。” 地支相冲
        # }

        month_str = []

        # 获取指定本卦指定爻对象
        yao_dizhi = guayao.dizhi #爻地支
        yao_wuxing = guayao.wuxing #爻五行

        #月建
        month_dizhi = self.month_dizhi #月建地支
        month_wuxing = self.month_wuxing #月建五行
       
        # 1. 优先判断月破（最高优先级，直接返回月破权重）
        if EarthlyBranch(month_dizhi).is_clashed_with(yao_dizhi):
            month_str.append("月破")
            base_force = YUE_JIAN_WEIGHTS["月破"]
        # 2. 判断临月建（值月）
        elif month_dizhi == yao_dizhi:
            month_str.append("临月建")
            base_force = YUE_JIAN_WEIGHTS["临月建"]
        # 3. 判断月合
        elif EarthlyBranch(month_dizhi).is_combined_with(yao_dizhi):
            month_str.append("月建合")
            base_force = YUE_JIAN_WEIGHTS["月建合"]
        # 4. 判断五行生克关系
        else:
            if FiveElements(month_wuxing).generates(yao_wuxing):
                month_str.append("月建生")
                base_force = YUE_JIAN_WEIGHTS["月建生"]
            elif FiveElements(month_wuxing).restrains(yao_wuxing):
                month_str.append("月建克")
                base_force = YUE_JIAN_WEIGHTS["月建克"]
            elif month_wuxing == yao_wuxing:
                month_str.append("月帮扶")
                base_force = YUE_JIAN_WEIGHTS["月帮扶"]
            else:
                month_str.append("月建无作用")
                base_force = YUE_JIAN_WEIGHTS["无作用"]
        
        # 5. 爻旬空 修正（空亡则力量减半）.月建不存在旬空，月统三旬
        if yao_dizhi in self.qigua_day_xunkong:
            month_str.append(f"{guayao.name} 爻空亡")
            yue_kong_coeff = 0.5
        else:
            yue_kong_coeff = 1

        wxxiusi = EarthlyBranch(month_dizhi).get_wangxiangxiuqiusi(yao_wuxing)

        guayao.sishiwangxiang = wxxiusi
        si_shi_coeff = SI_SHI[wxxiusi]
        
        # 7. 最终计算（基础力量 × 空亡系数 × 四时系数）
        final_force = base_force * yue_kong_coeff * si_shi_coeff
        guayao.month_score = (round(final_force, 4), month_str)
        print(f"月建对{guayao.name}爻力量计算（基础力量{base_force} × 空亡系数{yue_kong_coeff} × 四时系数{si_shi_coeff}) 最终得分:{guayao.month_score}")

        return (round(final_force, 4), month_str)

    def _calc_day_score(self, guayao: GuaYao) -> (float, list[str]):
        day_str = []

        # 获取指定本卦指定爻对象
        yao_dizhi = guayao.dizhi #爻地支
        yao_wuxing = guayao.wuxing #爻五行

        #日辰
        day_dizhi = self.day_dizhi #日辰地支
        day_wuxing = self.day_wuxing #日辰五行
        base_force = 0.0
        # 1. 优先判断临日建
        if day_dizhi == yao_dizhi:
            day_str.append("临日建")
            base_force = RI_CHEN_WEIGHTS["临日建"]
        elif yao_wuxing == day_wuxing:
            day_str.append("日辰帮扶比和")
            base_force = RI_CHEN_WEIGHTS["日帮扶"]
        elif EarthlyBranch(day_dizhi).is_combined_with(yao_dizhi):
            if guayao.dongyao:
                day_str.append("日辰合动爻，动爻行动力受限")
                base_force = RI_CHEN_WEIGHTS["日合动"]
            else:
                day_str.append("日辰合静爻，静爻变有力")
                base_force = RI_CHEN_WEIGHTS["日合静"]
        elif EarthlyBranch(day_dizhi).is_clashed_with(yao_dizhi):
            # 日辰冲 
            if guayao.sishiwangxiang in "旺相":
                day_str.append("日辰冲旺相静爻")    
                base_force = RI_CHEN_WEIGHTS["日冲"]
            else:
                day_str.append("日破")    
                base_force = RI_CHEN_WEIGHTS["日破"]
        elif FiveElements(day_wuxing).generates(yao_wuxing):
            day_str.append("日辰生")
            base_force = RI_CHEN_WEIGHTS["日生"]
        elif FiveElements(day_wuxing).restrains(yao_wuxing):
            day_str.append("日辰克")
            base_force = RI_CHEN_WEIGHTS["日克"]
        else:
            day_str.append("日辰无作用")
            base_force = RI_CHEN_WEIGHTS["无作用"]
        
        # 爻旬空 修正（空亡则力量减半）
        if yao_dizhi in self.qigua_day_xunkong:
            yao_kong_coeff = 0.5
        else:
            yao_kong_coeff = 1
        
        # 最终计算（基础力量 × 空亡系数）
        final_force = base_force * yao_kong_coeff
        guayao.day_score = (round(final_force, 4), day_str)
        print(f"日辰对{guayao.name}爻力量计算（基础力量{base_force} × 空亡系数{yao_kong_coeff} 最终得分:{guayao.day_score}")
        return (round(final_force, 4), day_str)

    def _calc_bianyao_coeff(self, guayao: GuaYao) -> (float, list[str]):
        '''
        1.硬判定优先：变爻若为月破、日破、旬空、入日月墓，直接有效系数 = 0，无生克权。
        2.量化日月分：变爻先算月建分 + 日辰分，得到变爻日月净分（S）。

            bianyao_str.append(f"{guayao.dizhi}{guayao.wuxing}变爻 旬空，系数为0，无生克权")
        elif EarthlyBranch.get_changsheng_stage(self.month_dizhi, guayao.wuxing) == "墓":
            bianyao_str.append(f"{guayao.dizhi}{guayao.wuxing}变爻 入月墓，系数为0，无生克权")
        elif EarthlyBranch.get_changsheng_stage(self.day_dizhi, guayao.wuxing) == "墓":
            bianyao_str.append(f"{guayao.dizhi}{guayao.wuxing}变爻 入日墓，系数为0，无生克权")
        
        if bianyao_str:
            return (0.0, bianyao_str)
        # 替换下一行
        # if EarthlyBranch(self.month_dizhi).is_clashed_with(guayao.dizhi) or \
        #     EarthlyBranch(self.day_dizhi).is_clashed_with(guayao.dizhi) or \

        # 2. 量化日月分
        month_score = self._calc_month_score(guayao)[0]
        day_score = self._calc_day_score(guayao)[0]
        S = month_score + day_score
        '''
        系数映射表（符合六爻旺衰理法）

        elif S >= 0:

        elif S >= -2:

        else:
            return 0.1

    def calc_yinyao_yongshen_score(self):
        '''
        应爻对用神的得分 及关系
        #### （一）核心判定维度（4 步走）
            1. 应爻是否为用神：应爻临用神，事在对方手中，对方态度直接决定成败。
            2. 应爻与用神的生克合冲：应爻生用神为助力，克用神为阻力，合为羁绊 / 助力，冲为变数。
            3. 应爻的旺衰与动静：旺相则助力 / 阻力强，休囚则弱；发动则影响即时，静爻则被动。
